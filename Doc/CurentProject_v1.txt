﻿using Asp.Versioning;
using Microsoft.Extensions.DependencyInjection;

namespace Crm.Api.Extensions;

public static class ApiVersioningExtensions
{
    public static IServiceCollection AddApiVersioningConfiguration(
        this IServiceCollection services)
    {
        services.AddApiVersioning(options =>
        {
            options.DefaultApiVersion = new ApiVersion(1.0);

            options.AssumeDefaultVersionWhenUnspecified = true;

            options.ReportApiVersions = true;
        })
        .AddApiExplorer(options =>
        {
            options.GroupNameFormat = "'v'VVV";

            options.SubstituteApiVersionInUrl = true;
        });

        return services;
    }
}

    protected readonly DbSet<TEntity> DbSet =
        context.Set<TEntity>();

    public async Task<TEntity?> GetByIdAsync(
        TKey id,
        CancellationToken cancellationToken = default)
    {
        return await DbSet.FindAsync(
            [id],
            cancellationToken);
    }

    public async Task<List<TEntity>> GetAllAsync(
        CancellationToken cancellationToken = default)
    {
        return await DbSet
            .Where(x => !x.IsDeleted)
            .ToListAsync(cancellationToken);
    }

    public async Task AddAsync(
        TEntity entity,
        CancellationToken cancellationToken = default)
    {
        await DbSet.AddAsync(
            entity,
            cancellationToken);
    }

    public void Update(TEntity entity)
    {
        DbSet.Update(entity);
    }

    public void Remove(TEntity entity)
    {
        entity.IsDeleted = true;
        DbSet.Update(entity);
    }
}

import os
import sys
from pathlib import Path

# ===== تنظیمات =====
# پوشه‌هایی که نباید اسکن شوند
EXCLUDE_DIRS = {
    'bin', 'obj', 'node_modules', 'venv', '.venv', '__pycache__',
    '.git', '.vs', '.vscode', 'packages', 'wwwroot', 'dist', 'build',
    'Migrations', 'migrations', 'Properties'
}

# پسوند فایل‌های کدی که محتوایشان نمایش داده شود
INCLUDE_EXTENSIONS = {
    '.cs', '.vb', '.fs', '.xaml', '.xml', '.json', '.config',
    '.cshtml', '.razor', '.js', '.ts', '.jsx', '.tsx',
    '.css', '.scss', '.sql', '.py', '.java', '.cpp', '.h'
}

# حداکثر اندازه فایل (بایت) - فایل‌های بزرگتر رد می‌شوند
MAX_FILE_SIZE = 1024 * 100  # 100 KB

# ============================================

def get_tree_structure(root_dir, prefix="", is_last=True):
    """ایجاد درخت پوشه به صورت متن"""
    lines = []
    items = sorted([p for p in root_dir.iterdir() if p.name not in EXCLUDE_DIRS])
    
    for i, item in enumerate(items):
        is_last_item = (i == len(items) - 1)
        connector = "└── " if is_last_item else "├── "
        
        if item.is_dir():
            lines.append(f"{prefix}{connector}{item.name}/")
            extension = "    " if is_last_item else "│   "
            lines.extend(get_tree_structure(item, prefix + extension, is_last_item))
        else:
            if item.suffix.lower() in INCLUDE_EXTENSIONS:
                lines.append(f"{prefix}{connector}{item.name}  📄")
            else:
                lines.append(f"{prefix}{connector}{item.name}")
    
    return lines

def collect_file_contents(root_dir):
    """جمع‌آوری محتوای فایل‌های کد"""
    contents = []
    
    for file_path in root_dir.rglob('*'):
        # بررسی پوشه‌های استثنا
        if any(excluded in file_path.parts for excluded in EXCLUDE_DIRS):
            continue
        
        if file_path.is_file():
            ext = file_path.suffix.lower()
            if ext in INCLUDE_EXTENSIONS:
                try:
                    size = file_path.stat().st_size
                    if size <= MAX_FILE_SIZE:
                        relative_path = file_path.relative_to(root_dir)
                        with open(file_path, 'r', encoding='utf-8') as f:
                            content = f.read()
                        contents.append({
                            'path': str(relative_path),
                            'content': content,
                            'ext': ext
                        })
                    else:
                        relative_path = file_path.relative_to(root_dir)
                        contents.append({
                            'path': str(relative_path),
                            'content': f'[فایل بزرگتر از {MAX_FILE_SIZE/1024:.0f}KB - محتوا نمایش داده نشد]',
                            'ext': ext
                        })
                except (UnicodeDecodeError, PermissionError):
                    # فایل باینری یا بدون دسترسی
                    relative_path = file_path.relative_to(root_dir)
                    contents.append({
                        'path': str(relative_path),
                        'content': '[فایل باینری یا قابل خواندن نیست]',
                        'ext': ext
                    })
    
    return contents

def generate_output(root_dir):
    """تولید فایل خروجی"""
    root_name = root_dir.name
    
    # بخش ۱: درخت پوشه
    output = []
    output.append("=" * 70)
    output.append(f"📁 ساختار پروژه: {root_name}")
    output.append("=" * 70)
    output.append("")
    output.extend(get_tree_structure(root_dir))
    output.append("")
    output.append("=" * 70)
    output.append("📄 محتوای فایل‌های کد")
    output.append("=" * 70)
    output.append("")
    
    # بخش ۲: محتوای فایل‌ها
    files = collect_file_contents(root_dir)
    
    # مرتب‌سازی بر اساس مسیر
    files.sort(key=lambda x: x['path'])
    
    for file_info in files:
        output.append("")
        output.append("-" * 70)
        output.append(f"📄 فایل: {file_info['path']}")
        output.append("-" * 70)
        output.append("")
        output.append(file_info['content'])
        output.append("")
    
    return "\n".join(output)

def main():
    # پوشه فعلی (جایی که اسکریپت اجرا می‌شود)
    root_dir = Path.cwd()
    
    # درخواست مسیر سفارشی (اختیاری)
    print(f"پوشه فعلی: {root_dir}")
    custom_path = input("برای استفاده از مسیر دیگر وارد کنید (یا Enter برای ادامه): ").strip()
    if custom_path:
        root_dir = Path(custom_path)
        if not root_dir.exists():
            print(f"❌ مسیر '{root_dir}' وجود ندارد!")
            return
    
    print(f"\n🔍 در حال اسکن: {root_dir}")
    
    # تولید خروجی
    output_text = generate_output(root_dir)
    
    # ذخیره در فایل
    output_file = root_dir / "project_structure.txt"

    print(f"📊 تعداد کل فایل‌های کد: {len([p for p in root_dir.rglob('*') if p.is_file() and p.suffix.lower() in INCLUDE_EXTENSIONS])}")
